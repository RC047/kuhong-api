[ Python ] : 
				
from requests import get
url = get('https://kuhong-api.herokuapp.com/status').json()
print(url)
[ PHP ] : 
				
$url = 'https://kuhong-api.herokuapp.com/status';
$curl = curl_init($url);
curl_setopt($curl, CURLOPT_RETURNTRANSFER, true);
curl_setopt($curl, CURLOPT_HTTPHEADER, [
  'Content-Type: application/json'
]);
$response = curl_exec($curl);
curl_close($curl);
echo $response . PHP_EOL;
[ Node JS ] : 
				
const axios = require('axios');
axios.get('https://kuhong-api.herokuapp.com/status')
  .then(res => {
  console.log(res.data);
});
.catch(err => {
  console.log(err)
})
